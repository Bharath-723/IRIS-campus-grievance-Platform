1. Show all raw submissions received
SELECT id, full_name, email, place, issue, severity_level, 
       issue_duration, wants_reward, submitted_at 
FROM raw_submissions
ORDER BY submitted_at DESC;


2. Show duplicate detection working
SELECT text_hash, COUNT(*) AS report_count, 
       MAX(issue) AS issue, MAX(place) AS place
FROM raw_submissions
GROUP BY text_hash
ORDER BY report_count DESC;


3. Show majority filter result — issues that crossed threshold
SELECT id, place, representative_issue, severity_level,
       frequency, first_raiser_name, status, created_at
FROM accepted_issues
ORDER BY frequency DESC;


4. Show severity wise issue count
SELECT severity_level, COUNT(*) AS total_issues
FROM accepted_issues
GROUP BY severity_level
ORDER BY 
    CASE severity_level
        WHEN 'Critical' THEN 1
        WHEN 'High'     THEN 2
        WHEN 'Medium'   THEN 3
        WHEN 'Low'      THEN 4
    END;


5. Show location wise issue count
SELECT place, COUNT(*) AS total_issues,
       SUM(frequency) AS total_reports
FROM accepted_issues
GROUP BY place
ORDER BY total_issues DESC;


6. Show solutions submitted per issue
SELECT ai.representative_issue, ai.place,
       COUNT(s.id) AS total_solutions,
       SUM(CASE WHEN s.wants_reward = 1 THEN 1 ELSE 0 END) AS reward_seekers
FROM accepted_issues ai
LEFT JOIN solutions s ON s.issue_id = ai.id
GROUP BY ai.representative_issue, ai.place
ORDER BY total_solutions DESC;


7. Show status wise breakdown
SELECT status, COUNT(*) AS count
FROM accepted_issues
GROUP BY status;


8. Show how many duplicates were removed
SELECT 
    COUNT(*) AS total_submissions,
    SUM(CASE WHEN is_duplicate = 0 THEN 1 ELSE 0 END) AS unique_issues,
    SUM(CASE WHEN is_duplicate = 1 THEN 1 ELSE 0 END) AS duplicates_removed
FROM raw_submissions;


9. Show full issue pipeline — from submission to acceptance
SELECT 
    rs.full_name, rs.email, rs.place,
    rs.issue, rs.severity_level,
    rs.is_duplicate,
    CASE WHEN ai.id IS NOT NULL THEN 'Accepted' ELSE 'Pending' END AS pipeline_status
FROM raw_submissions rs
LEFT JOIN accepted_issues ai 
    ON CAST(ai.representative_issue AS VARCHAR(MAX)) = CAST(rs.issue AS VARCHAR(MAX))
ORDER BY pipeline_status DESC;


10. Show reward contestants
SELECT s.submitter_name, s.submitter_email,
       ai.representative_issue, ai.place,
       s.solution_text, s.wants_reward
FROM solutions s
JOIN accepted_issues ai ON ai.id = s.issue_id
WHERE s.wants_reward = 1;
